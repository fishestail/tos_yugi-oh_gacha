    <script>
        const POOL = [
            { id: 1, name: "武藤遊戲＆破壞龍鋼多拉", star: 6, type: "grand", normalRate: 0.01, boostedRate: 0.02, img: "images/yugi.png.webp" },
            { id: 2, name: "闇遊戲＆黑魔導", star: 6, type: "grand", normalRate: 0.01, boostedRate: 0.02, img: "images/dark_yugi.png.webp" },
            { id: 3, name: "海馬瀨人＆青眼白龍", star: 6, type: "rare", normalRate: 0.045, boostedRate: 0.10, img: "images/kaiba.png.webp" },
            { id: 4, name: "城之內克也＆真紅眼黑龍", star: 6, type: "rare", normalRate: 0.045, boostedRate: 0.10, img: "images/jonouchi.png.webp" },
            { id: 5, name: "闇獏良＆暗黑人偶・妮可羅菲婭", star: 6, type: "normal", normalRate: 0.29666, boostedRate: 0.25333, img: "images/bakura.png.webp" },
            { id: 6, name: "孔雀舞＆神鷹女郎三姐妹", star: 6, type: "normal", normalRate: 0.29666, boostedRate: 0.25333, img: "images/mai.png.webp" },
            { id: 7, name: "伊西絲 ‧ 伊修達爾", star: 6, type: "normal", normalRate: 0.29666, boostedRate: 0.25333, img: "images/ishizu.png.webp" }
        ];

        let isBoosted = false;
        let totalDraws = 0;
        let pityCount = 35;
        let inventory = {};
        let historyLogs = [];

        function toggleRate() {
            isBoosted = !isBoosted;
            const btn = document.getElementById("rateToggleBtn");
            btn.classList.toggle("active", isBoosted);
            btn.innerText = isBoosted ? "加倍中 🔥" : "未加倍";
            renderCardList();
        }

        function renderCardList() {
            const listEl = document.getElementById("cardList");
            listEl.innerHTML = POOL.map(card => {
                const rate = (isBoosted ? card.boostedRate : card.normalRate) * 100;
                return `
                    <div class="card-row ${card.type}">
                        <img src="${card.img}" alt="${card.name}">
                        <div class="card-info">
                            <div class="card-name">${card.star}★ ${card.name}</div>
                            <div class="card-rate">${rate.toFixed(3).replace(/\\.?0+$/, '')}%</div>
                        </div>
                    </div>
                `;
            }).join("");
        }

        function pullSingle() {
            totalDraws++;
            pityCount--;

            let drawnCard = null;

            if (pityCount <= 0) {
                const grandList = POOL.filter(c => c.type === "grand");
                drawnCard = grandList[Math.floor(Math.random() * grandList.length)];
                pityCount = 35;
            } else {
                const rand = Math.random();
                let acc = 0;
                for (let card of POOL) {
                    acc += isBoosted ? card.boostedRate : card.normalRate;
                    if (rand <= acc) {
                        drawnCard = card;
                        break;
                    }
                }
                if (!drawnCard) drawnCard = POOL[POOL.length - 1];

                if (drawnCard.type === "grand") {
                    pityCount = 35;
                }
            }

            inventory[drawnCard.name] = (inventory[drawnCard.name] || 0) + 1;

            historyLogs.unshift({
                index: totalDraws,
                card: drawnCard
            });

            return drawnCard;
        }

        function drawCards(times) {
            let sessionPulls = [];
            for (let i = 0; i < times; i++) {
                sessionPulls.push(pullSingle());
            }
            updateHUD();
            renderInventory();
            renderHistory();
            showModal(sessionPulls);
        }

        function updateHUD() {
            document.getElementById("totalDraws").innerText = totalDraws;
            document.getElementById("totalStones").innerText = totalDraws * 5;
            document.getElementById("pityCount").innerText = pityCount;
        }

        function renderInventory() {
            const grid = document.getElementById("inventoryGrid");
            grid.innerHTML = "";

            POOL.forEach(card => {
                const count = inventory[card.name] || 0;
                if (count > 0) {
                    grid.innerHTML += `
                        <div class="card-tile ${card.type}">
                            <div class="badge">x${count}</div>
                            <img src="${card.img}" alt="${card.name}">
                            <div class="title">${card.name}</div>
                        </div>
                    `;
                }
            });
        }

        function renderHistory() {
            const list = document.getElementById("historyList");
            list.innerHTML = historyLogs.slice(0, 50).map(log => `
                <div class="history-item">
                    <span>第 ${log.index} 抽</span>
                    <span style="color: ${log.card.type === 'grand' ? 'var(--gold)' : (log.card.type === 'rare' ? 'var(--blue)' : '#cbd5e1')}">
                        ${log.card.name}
                    </span>
                </div>
            `).join("");
        }

        function showModal(cards) {
            const modal = document.getElementById("drawModal");
            const grid = document.getElementById("modalGrid");
            document.getElementById("modalTitle").innerText = cards.length > 1 ? "🎉 十連抽結果" : "🎉 抽卡結果";
            
            grid.innerHTML = cards.map(card => `
                <div class="modal-card ${card.type}">
                    <img src="${card.img}" alt="${card.name}">
                    <div class="name">${card.name}</div>
                </div>
            `).join("");

            modal.style.display = "flex";
        }

        function closeModal() {
            document.getElementById("drawModal").style.display = "none";
        }

        function switchView(view) {
            const bagGrid = document.getElementById("inventoryGrid");
            const historyList = document.getElementById("historyList");
            const tabBag = document.getElementById("tabBag");
            const tabHistory = document.getElementById("tabHistory");

            if (view === 'bag') {
                bagGrid.style.display = "grid";
                historyList.style.display = "none";
                tabBag.classList.add("active");
                tabHistory.classList.remove("active");
            } else {
                bagGrid.style.display = "none";
                historyList.style.display = "flex";
                tabBag.classList.remove("active");
                tabHistory.classList.add("active");
            }
        }

        function resetAll() {
            if (confirm("確定要重置所有抽卡數據嗎？")) {
                totalDraws = 0;
                pityCount = 35;
                inventory = {};
                historyLogs = [];
                updateHUD();
                renderInventory();
                renderHistory();
            }
        }

        renderCardList();
    </script>
